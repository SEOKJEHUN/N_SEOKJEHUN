def calculate_summary():
    """Firestore 'data' 컬렉션에서 데이터를 읽어와 통계 및 요약 정보를 생성"""
    docs = db.collection("data").stream()
    data_list = []
    
    for doc in docs:
        item = doc.to_dict()
        item["id"] = doc.id
        data_list.append(item)
        
    count = len(data_list)
    if count == 0:
        return {
            "period": "데이터 없음",
            "count": 0,
            "metrics": {"total": 0, "average": 0, "max": 0, "min": 0},
            "trend": "데이터가 없습니다."
        }
    
    # date 기준 정렬
    data_list.sort(key=lambda x: x.get("date", ""))
    
    values = [float(x.get("value", 0)) for x in data_list]
    total = sum(values)
    average = round(total / count, 2)
    max_val = max(values)
    min_val = min(values)
    
    period = f"{data_list[0].get('date')} ~ {data_list[-1].get('date')}"
    
    # 간단 추세 계산 (최근 3개 평균 vs 전체 평균 비교 등)
    trend = "유지 중"
    if count >= 3:
        recent_avg = sum(values[-3:]) / 3
        if recent_avg > average:
            trend = "상승 추세"
        elif recent_avg < average:
            trend = "하락 추세"

    return {
        "period": period,
        "count": count,
        "metrics": {
            "total": total,
            "average": average,
            "max": max_val,
            "min": min_val
        },
        "trend": trend
    }

import os

from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from typing import Optional, List
from openai import OpenAI
from dotenv import load_dotenv

from service import db, calculate_summary

load_dotenv()

app = FastAPI(title="Data-Aware AI Assistant API", version="1.0.0")

# CORS 설정
origins = os.getenv("ALLOWED_ORIGINS", "*").split(",")
app.add_middleware(
    CORSMiddleware,
    allow_origins=origins,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

openai_client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

# Pydantic 모델
class DataItem(BaseModel):
    date: str
    value: float
    memo: Optional[str] = ""

class ChatRequest(BaseModel):
    message: str
    conversation_id: Optional[str] = None

class ConversationSave(BaseModel):
    title: str
    messages: List[dict]


# --- 1. Data CRUD & Summary API ---

@post("/api/data", status_code=status.HTTP_201_CREATED)
def create_data(item: DataItem):
    try:
        doc_ref = db.collection("data").document()
        doc_ref.set(item.model_dump())
        return {"id": doc_ref.id, **item.model_dump()}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@get("/api/data")
def get_data_list():
    docs = db.collection("data").stream()
    result = []
    for doc in docs:
        item = doc.to_dict()
        item["id"] = doc.id
        result.append(item)
    return result

@put("/api/data/{item_id}")
def update_data(item_id: str, item: DataItem):
    doc_ref = db.collection("data").document(item_id)
    if not doc_ref.get().exists:
        raise HTTPException(status_code=404, detail="Data not found")
    doc_ref.update(item.model_dump())
    return {"id": item_id, **item.model_dump()}

@delete("/api/data/{item_id}")
def delete_data(item_id: str):
    doc_ref = db.collection("data").document(item_id)
    if not doc_ref.get().exists:
        raise HTTPException(status_code=404, detail="Data not found")
    doc_ref.delete()
    return {"message": "Successfully deleted", "id": item_id}

@get("/api/data/summary")
def get_data_summary():
    return calculate_summary()

@post("/api/chat")
def chat_with_ai(req: ChatRequest):
    # 1. 데이터 요약 정보 가져오기
    summary = calculate_summary()
    
    # 2. 시스템 프롬프트 구성 (컨텍스트 주입)
    system_prompt = f"""당신은 사용자 데이터를 완벽히 이해하고 분석해주는 전문 AI 비서입니다.

[사용자 데이터 요약]
- 데이터 기간: {summary['period']}
- 총 레코드 수: {summary['count']}개
- 주요 지표: 총합 {summary['metrics']['total']}, 평균 {summary['metrics']['average']}, 최대 {summary['metrics']['max']}, 최소 {summary['metrics']['min']}
- 최근 트렌드: {summary['trend']}

위 요약 데이터를 기반으로 사용자의 질문에 친절하고 정확하게 답변해주세요. 데이터에 근거해 구체적인 숫자를 인용해 설명하는 것이 좋습니다."""

    try:
        response = openai_client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": req.message}
            ],
            temperature=0.7
        )
        answer = response.choices[0].message.content
        
        # 대화 기록 자동 저장 (선택 사항 혹은 즉시 반영)
        return {
            "reply": answer,
            "summary_used": summary
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"OpenAI API Error: {str(e)}")

// 환경에 맞게 API Base URL 변경 (로컬 테스트 시 http://localhost:8000)
const API_BASE_URL = window.location.hostname === "localhost" ? "http://localhost:8000" : "https://your-backend-service.onrender.com";

let currentMessages = [];

window.onload = function() {
    loadSummary();
    loadDataList();
    loadConversations();
};

function switchTab(tabName) {
    document.querySelectorAll('.tab-section').forEach(sec => sec.style.display = 'none');
    document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
    
    if (tabName === 'chat') {
        document.getElementById('chat-section').style.display = 'flex';
        event.target.classList.add('active');
    } else {
        document.getElementById('data-section').style.display = 'flex';
        event.target.classList.add('active');
        loadDataList();
    }
}

// --- 요약 정보 조회 ---
async function loadSummary() {
    try {
        const res = await fetch(`${API_BASE_URL}/api/data/summary`);
        const data = await res.json();
        const box = document.getElementById('summary-box');
        box.innerHTML = `
            <p><b>기간:</b> ${data.period}</p>
            <p><b>총 건수:</b> ${data.count}개</p>
            <p><b>평균:</b> ${data.metrics.average.toLocaleString()}</p>
            <p><b>최고/최저:</b> ${data.metrics.max.toLocaleString()} / ${data.metrics.min.toLocaleString()}</p>
            <p><b>트렌드:</b> ${data.trend}</p>
        `;
    } catch (err) {
        console.error("Summary load error:", err);
    }
}

// --- 채팅 기능 ---
async function sendMessage() {
    const input = document.getElementById('user-input');
    const text = input.value.trim();
    if (!text) return;

    appendMessage(text, 'user');
    currentMessages.push({ role: 'user', content: text });
    input.value = '';

    // 로딩 표시 추가
    const chatWindow = document.getElementById('chat-window');
    const loadingId = 'loading-' + Date.now();
    const loadingDiv = document.createElement('div');
    loadingDiv.className = 'message ai';
    loadingDiv.id = loadingId;
    loadingDiv.innerText = 'AI가 데이터를 분석하며 답변을 생성 중입니다...';
    chatWindow.appendChild(loadingDiv);
    chatWindow.scrollTop = chatWindow.scrollHeight;

    try {
        const res = await fetch(`${API_BASE_URL}/api/chat`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message: text })
        });
        const data = await res.json();
        
        // 로딩 제거 후 답변 추가
        document.getElementById(loadingId).remove();
        appendMessage(data.reply, 'ai');
        currentMessages.push({ role: 'assistant', content: data.reply });

        // 대화 자동 저장 (선택 사항)
        saveCurrentConversation(text, data.reply);
    } catch (err) {
        document.getElementById(loadingId).remove();
        appendMessage('오류가 발생했습니다. 백엔드 연결을 확인해주세요.', 'ai');
    }
}

function appendMessage(text, sender) {
    const chatWindow = document.getElementById('chat-window');
    const div = document.createElement('div');
    div.className = `message ${sender}`;
    div.innerText = text;
    chatWindow.appendChild(div);
    chatWindow.scrollTop = chatWindow.scrollHeight;
}

function handleKeyPress(e) {
    if (e.key === 'Enter') sendMessage();
}

// --- 데이터 관리 (CRUD) ---
async function loadDataList() {
    try {
        const res = await fetch(`${API_BASE_URL}/api/data`);
        const list = await res.json();
        const tbody = document.getElementById('data-tbody');
        tbody.innerHTML = '';
        
        list.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td>${item.date}</td>
                <td>${Number(item.value).toLocaleString()}</td>
                <td>${item.memo || ''}</td>
                <td><button onclick="deleteData('${item.id}')" style="background:#e53e3e; color:white; border:none; padding:4px 8px; border-radius:4px; cursor:pointer;">삭제</button></td>
            `;
            tbody.appendChild(tr);
        });
    } catch (err) {
        console.error("Data load error:", err);
    }
}

async function handleDataSubmit(e) {
    e.preventDefault();
    const date = document.getElementById('data-date').value;
    const value = parseFloat(document.getElementById('data-value').value);
    const memo = document.getElementById('data-memo').value;

    try {
        await fetch(`${API_BASE_URL}/api/data`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ date, value, memo })
        });
        document.getElementById('data-form').reset();
        loadDataList();
        loadSummary();
    } catch (err) {
        alert('데이터 추가 실패');
    }
}

async function deleteData(id) {
    if (!confirm('정말 삭제하시겠습니까?')) return;
    try {
        await fetch(`${API_BASE_URL}/api/data/${id}`, { method: 'DELETE' });
        loadDataList();
        loadSummary();
    } catch (err) {
        alert('삭제 실패');
    }
}

// --- 대화 기록 ---
async function saveCurrentConversation(firstQ, firstA) {
    try {
        await fetch(`${API_BASE_URL}/api/conversations`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                title: firstQ.length > 15 ? firstQ.substring(0, 15) + '...' : firstQ,
                messages: [{ role: 'user', content: firstQ }, { role: 'assistant', content: firstA }]
            })
        });
        loadConversations();
    } catch (err) {
        console.error("Conversation save error:", err);
    }
}

async function loadConversations() {
    try {
        const res = await fetch(`${API_BASE_URL}/api/conversations`);
        const list = await res.json();
        const ul = document.getElementById('conv-list');
        ul.innerHTML = '';
        list.forEach(conv => {
            const li = document.createElement('li');
            li.innerText = conv.title || '대화 기록';
            li.onclick = () => loadConversationDetail(conv.id);
            ul.appendChild(li);
        });
    } catch (err) {
        console.error("Conversations list load error:", err);
    }
}

async function loadConversationDetail(id) {
    try {
        const res = await fetch(`${API_BASE_URL}/api/conversations/${id}`);
        const data = await res.json();
        const chatWindow = document.getElementById('chat-window');
        chatWindow.innerHTML = '';
        data.messages.forEach(m => {
            appendMessage(m.content, m.role === 'user' ? 'user' : 'ai');
        });
        switchTab('chat');
    } catch (err) {
        alert('대화 불러오기 실패');
    }
}

# 🤖 내 데이터를 아는 AI 비서 (Data-Aware AI Assistant)

일반적인 ChatGPT가 알지 못하는 **개인/비즈니스 시계열 데이터**를 Firestore에 연동하고, 데이터 요약 정보를 시스템 프롬프트에 동적으로 주입(Context Injection)하여 맞춤형 답변을 제공하는 웹 서비스입니다.

---

## 🚀 기술 스택
- **Backend:** Python, FastAPI, Uvicorn, OpenAI API, Firebase Admin (Firestore)
- **Frontend:** Vanilla HTML5, CSS3, JavaScript (ES6)
- **Deployment:** Render (백엔드), Vercel (프론트엔드)

---
